                        <svg viewBox="0 0 32 10" className={`h-3 w-auto transition-colors duration-300 ${paymentMethod === 'paymob' || paymentMethod === 'card' ? 'fill-white' : 'fill-white/30'}`} xmlns="http://www.w3.org/2000/svg">

                        </svg>

                  <div className="absolute top-3 right-3 opacity-0 peer-checked:opacity-100 transition-opacity duration-300 text-[#e63946]">

            {/* Card Details Form (Dynamically loads Paymob/PayPal/Stripe) */}
            <section className="bg-white/5 border border-white/10 rounded-2xl p-4">
              <h2 className="text-xl font-semibold mb-4">بيانات الدفع</h2>
              <div className="w-full">
                {(paymentMethod === 'paymob' || paymentMethod === 'card') && (
                  <div className="w-full min-h-[250px]">

                  window.dispatchEvent(new Event('payFromOutside'));

              >
                <span className="material-symbols-outlined" style={{ fontVariationSettings: "'FILL' 1" }}>lock</span>
                <span className="text-sm font-medium">إتمام الدفع - {formatPriceShort(selectedPlanForPay)}</span>
              </button>
              
              <button 
                onClick={() => {
                  setSelectedPlanForPay(null);
                  setIsProcessingPayment(false);
                }}
                className="flex items-center justify-center text-white/60 px-4 py-2 hover:bg-white/10 rounded-lg transition-colors"
              >
                <span className="material-symbols-outlined mb-1 block md:hidden">close</span>
                <span className="text-sm font-medium">إلغاء</span>
              </button>
            </nav>
          )}
        </div>
      )}
    </div>
  );
